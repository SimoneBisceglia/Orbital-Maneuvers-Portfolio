function [torque_real, rw_power] = reaction_wheels_model(torque_cmd, rw_max_torque, rw_power_W_per_Nm)
    % =====================================================================
    % REACTION WHEELS ACTUATOR MODEL
    % Simulates physical torque saturation and power consumption.
    % =====================================================================
    
    torque_real = zeros(3,1);
    
    % 1. TORQUE SATURATION (Hardware limits)
    % The wheels cannot provide more than 'rw_max_torque' per axis.
    for i = 1:3
        if torque_cmd(i) > rw_max_torque
            torque_real(i) = rw_max_torque;
        elseif torque_cmd(i) < -rw_max_torque
            torque_real(i) = -rw_max_torque;
        else
            torque_real(i) = torque_cmd(i);
        end
    end
    
    % PHYSICAL & MODELING ARCHITECTURE NOTE:
    % In this quasi-static architecture, 'torque_real' is directly fed into 
    % the standard rigid-body Euler equations as an external body torque input, 
    % where it is summed with environmental disturbances:
    %   I*omega_dot + omega x (I*omega) = tau_rw + tau_env
    % 
    % Physically, the reaction wheel torque is generated by the internal 
    % change of the wheel's angular momentum (tau_rw = -dot(h_rw)). By directly 
    % summing the torques without explicitly moving dot(h_rw) to the LHS of 
    % the dynamics equation, we omit tracking the internal momentum state (h_rw).
    % 

    % CONSEQUENCE & LONG-DURATION MISSIONS:
    % Because h_rw is not integrated here, this model accurately captures 
    % *torque saturation* (rw_max_torque), but it does *not* simulate wheel 
    % speed/momentum saturation. For multi-day simulations (e.g., 6 days), 
    % continuous environmental disturbances (tau_env) would accumulate in 
    % h_rw, requiring an explicit momentum-integration loop and a magnetorquer 
    % unloading (dumping) logic to prevent unmodeled rotor speed overflow.
    
    % 2. POWER CONSUMPTION BUDGET [Watts]
    % Simplified model: Power is linearly proportional to the torque provided.
    rw_power = sum(abs(torque_real)) * rw_power_W_per_Nm;
    
end

    function [torque_real, mtq_power] = magnetorquer_model(torque_cmd, t, omega_orb_mag, mtq_max_dipole, mtq_power_W_per_Am2)
    % =====================================================================
    % MAGNETORQUER ACTUATOR MODEL
    % Implements the Cross-Product Control Law and physical limitations.
    % =====================================================================
    
    % 1. MOCK EARTH MAGNETIC FIELD (B) IN BODY FRAME [Tesla]
    % For this trade-off, we simulate a realistic B-field that oscillates 
    % as the satellite travels along its orbit.
    B_mag = 40e-6; % ~40 microTesla in LEO
    B_body = B_mag * [sin(omega_orb_mag * t); cos(omega_orb_mag * t); 0.5];
    
    % 2. IDEAL DIPOLE COMMAND (Cross-Product Law)
    % Formula: m = (B x tau_cmd) / |B|^2
    % This calculates the required dipole to approximate the requested torque.
    m_cmd = cross(B_body, torque_cmd) / (norm(B_body)^2 + 1e-12);
    
    % 3. SATURATION (Hardware limits)
    m_real = zeros(3,1);
    for i = 1:3
        if m_cmd(i) > mtq_max_dipole
            m_real(i) = mtq_max_dipole;
        elseif m_cmd(i) < -mtq_max_dipole
            m_real(i) = -mtq_max_dipole;
        else
            m_real(i) = m_cmd(i);
        end
    end
    
    % 4. REAL TORQUE GENERATED
    % The actual torque applied to the spacecraft: tau = m x B
    torque_real = cross(m_real, B_body);
    
    % 5. POWER CONSUMPTION BUDGET [Watts]
    mtq_power = sum(abs(m_real)) * mtq_power_W_per_Am2;
end 

    function [torque_real, mass_flow_rate] = thrusters_model(torque_cmd, thruster_max_torque, thruster_arm, thruster_Isp, g0)
    % =====================================================================
    % THRUSTER ACTUATOR MODEL (BANG-BANG CONTROL)
    % Simulates ON/OFF firing logic and propellant mass flow rate.
    % =====================================================================
    
    torque_real = zeros(3,1);
    
    % Deadband to prevent infinite firing for microscopic errors
    deadband = 0.005; % [Nm]
    
    % 1. BANG-BANG LOGIC
    for i = 1:3
        if torque_cmd(i) > deadband
            torque_real(i) = thruster_max_torque;
        elseif torque_cmd(i) < -deadband
            torque_real(i) = -thruster_max_torque;
        else
            torque_real(i) = 0.0;
        end
    end
    
    % 2. PROPELLANT CONSUMPTION BUDGET [kg/s]
    % Force = Torque / Arm. Mass flow = Force / (Isp * g0)
    total_force = sum(abs(torque_real)) / thruster_arm;
    mass_flow_rate = total_force / (thruster_Isp * g0);
    
end
